local function detectExecutor()
    if type(syn)=="table" and syn.protect_gui then
        return "Synapse X"
    elseif type(pebc_execute)=="function" or _G.krnl then
        return "Krnl"
    elseif type(is_sirhurt_closure)=="boolean" then
        return "Sirhurt"
    elseif type(secure_load)=="function" then
        return "ScriptWare"
    elseif type(getexecutorname)=="function" then
        return getexecutorname()
    elseif type(getexecutor)=="function" then
        return getexecutor()
    else
        return "Unknown"
    end
end

-- AIRPLANE‐STYLE UNDERGROUND TP
local function undergroundTeleport(cf)
    if not hrp then return end
    local steps=20
    local drop=undergroundDepth
    local tiltTotal=math.rad(-90)
    local tiltStep=tiltTotal/steps
    local dropStep=drop/steps
    for i=1,steps do
        hrp.CFrame = hrp.CFrame
            * CFrame.new(0,-dropStep,0)
            * CFrame.Angles(tiltStep,0,0)
        task.wait(0.03)
    end
    hrp.CFrame = cf
end

-- NEW: Toggle Invisible/No-Collide Doors
local function toggleDoorsInvisible(state)
    if doorsInvisibleConn then doorsInvisibleConn:Disconnect() doorsInvisibleConn = nil end
    
    local function setDoorProperties(part, trans, collide)
        if part:IsA("BasePart") then
            part.Transparency = trans
            part.CanCollide = collide
        end
    end
    
    local function applyToAllDoors(trans, collide)
        pcall(function()
            -- Main doors
            local doorsFolder = Workspace:FindFirstChild("Doors")
            if doorsFolder then
                for _, door in ipairs(doorsFolder:GetDescendants()) do
                    setDoorProperties(door, trans, collide)
                end
            end
            -- Cell doors
            local cells = Workspace:FindFirstChild("Prison Cells")
            if cells then
                for _, cell in ipairs(cells:GetChildren()) do
                    local door = cell:FindFirstChild("door")
                    if door then
                        setDoorProperties(door, trans, collide)
                        -- If door is a model, apply to its parts too
                        for _, part in ipairs(door:GetDescendants()) do
                            setDoorProperties(part, trans, collide)
                        end
                    end
                end
            end
        end)
    end
    
    if state then
        applyToAllDoors(1, false)  -- Invisible and no-collide
        doorsInvisibleConn = RunService.Heartbeat:Connect(function()
            applyToAllDoors(1, false)  -- Persist in case game resets properties
        end)
    else
        applyToAllDoors(0, true)  -- Reset to normal
    end
end

-- Fly (PC‐only)
local function toggleFly(state)
    if not isPC then
        if Toggles.Fly then Toggles.Fly:Set(false) end
        return
    end
    if flyConn then flyConn:Disconnect() flyConn=nil end
    
    -- FIXED: Safe nil-checks for cleanup
    local bv = hrp:FindFirstChild("BodyVelocity")
    if bv then bv:Destroy() end
    local bg = hrp:FindFirstChild("BodyGyro")
    if bg then bg:Destroy() end

    if not state then return end

    bg=Instance.new("BodyGyro",hrp)
    bg.MaxTorque=Vector3.new(9e9,9e9,9e9); bg.P=9e4
    bv=Instance.new("BodyVelocity",hrp)
    bv.MaxForce=Vector3.new(9e9,9e9,9e9); bv.Velocity=Vector3.new(0,0.1,0)

    flyConn = RunService.Heartbeat:Connect(function()
        if not Toggles.Fly.Value then return toggleFly(false) end
        local cam=Workspace.CurrentCamera
        local move=Vector3.new()

        if isPC then
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then move+=cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then move-=cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then move-=cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then move+=cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move+=Vector3.yAxis end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move-=Vector3.yAxis end
        else
            local md=hum.MoveDirection
            local fwd=Vector3.new(cam.CFrame.LookVector.X,0,cam.CFrame.LookVector.Z).Unit
            local rgt=Vector3.new(cam.CFrame.RightVector.X,0,cam.CFrame.RightVector.Z).Unit
            move = fwd*md.Z + rgt*md.X
            if mobileUp then move+=Vector3.yAxis; mobileUp=false end
        end

        bv.Velocity = (move.Magnitude>0 and move.Unit*flySpeed or Vector3.new(0,0.1,0))
        bg.CFrame = cam.CFrame
    end)
end

-- Car Fly
local function toggleCarFly(state)
    if carFlyConn then carFlyConn:Disconnect() carFlyConn=nil end
    if not state then return end

    carFlyConn = RunService.Heartbeat:Connect(function()
        if not Toggles.CarFly.Value then return end
        local seat = char:FindFirstChildOfClass("VehicleSeat")
        if not seat then return end

        local cam=Workspace.CurrentCamera
        local move=Vector3.new()
        if isPC then
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then move+=cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then move-=cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then move-=cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then move+=cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move+=Vector3.yAxis end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move-=Vector3.yAxis end
        else
            local md=hum.MoveDirection
            local fwd=Vector3.new(cam.CFrame.LookVector.X,0,cam.CFrame.LookVector.Z).Unit
            local rgt=Vector3.new(cam.CFrame.RightVector.X,0,cam.CFrame.RightVector.Z).Unit
            move = fwd*md.Z + rgt*md.X
            if mobileUp then move+=Vector3.yAxis; mobileUp=false end
        end

        local vehicle=seat.Parent
        local root=vehicle.PrimaryPart or seat
        local bv = root:FindFirstChild("BodyVelocity") or Instance.new("BodyVelocity",root)
        bv.MaxForce=Vector3.new(1/0,1/0,1/0)
        bv.Velocity=move.Unit*carFlySpeed
        vehicle:SetPrimaryPartCFrame(CFrame.new(root.Position)*cam.CFrame.Rotation)
    end)
end

-- Godmode
local function toggleGod(state)
    if godConn then godConn:Disconnect() godConn=nil end
    if state then
        godConn = hum.HealthChanged:Connect(function(h)
            if h < hum.MaxHealth then hum.Health=hum.MaxHealth end
        end)
    end
end

-- Invisibility
local function toggleInvis(state)
    if invisConn then invisConn:Disconnect() invisConn=nil end
    if state then
        invisConn = RunService.Heartbeat:Connect(function()
            for _,o in char:GetDescendants() do
                if o:IsA("BasePart") or o:IsA("Texture") or o:IsA("Decal") then
                    o.Transparency=1
                end
            end
        end)
    else
        for _,o in char:GetDescendants() do
            if o:IsA("BasePart") or o:IsA("Texture") or o:IsA("Decal") then
                o.Transparency=0
            end
        end
    end
end

-- Fast Shoot
local function toggleFastShoot(state)
    if fastShootConn then fastShootConn:Disconnect() fastShootConn=nil end
    if state then
        fastShootConn = RunService.Heartbeat:Connect(function()
            local tool = char:FindFirstChildOfClass("Tool")
            if tool then
                local gs = tool:FindFirstChild("GunScript") or tool:FindFirstChildOfClass("LocalScript")
                if gs and gs:FindFirstChild("Cooldown") then gs.Cooldown.Value=0.05 end
            end
        end)
    end
end

-- Rainbow Gun
local function toggleRainbowGun(state)
    if rainbowConn then rainbowConn:Disconnect() rainbowConn=nil end
    if state then
        rainbowConn = RunService.Heartbeat:Connect(function()
            local tool = char:FindFirstChildOfClass("Tool")
            if tool then
                local hue = tick()%5/5
                for _,p in tool:GetDescendants() do
                    if p:IsA("BasePart") then p.Color=Color3.fromHSV(hue,1,1) end
                end
            end
        end)
    end
end
